##  Ansible — Setup Nginx with Reverse Proxy

-name: Setup Nginx with Reverse Proxy
 hosts: all 
 become: yes
 tasks: 
   - name: Inginx 
     state: present 

   - name: Configure nginx as proxy
     template:
        src: reverse-proxy.conf.j2
        dest: /etc/nginx/conf.d/reverse-proxy-conf
   - nanme: Restart nginx service
     service:
     name: nginx
     state: restarted
     enabled: yes

#Kubernetes - Ingress with Nginx Controller
apiVersion : networking.k8s.io/v1
metadata:
   name: pathnex-ingress
   annotations:
       nginx.ingress.kubernetes.io/rewrite: /
  spec:
    rules:
      - host: pathnex.local
       http:
       - path: /
       pathType: Prefix
          backend:
            service:
              name: pathnex-service
              port:
                number: 80

GitLab CI/CD — Multi-Stage Deployment with Helm
stages:
  - build
  - push
  - deploy

build:
  stage: build
  script:
    - docker build -t pathnex-web-app .

push:
  stage: push
  script:
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push pathnex-web-app

deploy:
  stage: deploy
  script:
    - helm upgrade --install pathnex-nginx pathnex/nginx-ingress
