#Ansible

-- name: Setup Nginx with Custum config 
   hosts: all 
   become: yes 

   tasks:
   - name: Install nginx 
     state present
   - name : Copy files host to remote 
     copy:
        src: nginx.conf
        dest: /etc/nginx/nginx.config
   - name: Start nginx service
     service: 
     name: nginx
     state: started
     enabled: yes
