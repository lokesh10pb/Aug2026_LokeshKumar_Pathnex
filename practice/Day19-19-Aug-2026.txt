---
- name: Install Prometheus for Monitoring
  hosts: all
  become: yes

  tasks:
    - name: Install Prometheus packages
      yum:
        name:
          - prometheus
          - prometheus-node-exporter
        state: present

    - name: Start Prometheus service
      service:
        name: prometheus
        state: started
        enabled: yes

    - name: Start Prometheus Node Exporter
      service:
        name: prometheus-node-exporter
        state: started
        enabled: yes
