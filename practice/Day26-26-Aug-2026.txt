
#Ansible
- name: Implementing Ssh key-bases authorized_key
  hosts: all
  become: yes 
  tasks:
  - name: Generate key pair
    command: ssh-keygen -t rsa -b 2048 -f /home/user/.ssh/id_rsa -N
  - name: Add SSH public key baes authorized_keys
    authorized_key:
    user: user
    state: present
    key: "{{ lookup('file', '/home/user/.ssh/id_rsa.pub') }}"
