# Ansinle for installing in docker on remote server 

---
- name: Install docker
  hosts: all 
  become: yes


  tasks:
    - name: installing docker
      yum:
         name: docker
         state: present
         enabled: yes
    - name: starting docker
      service:
         name:docker
         state: started
         enabled: yes

   volumeMounts:

  - name: html-volume
    hostPath:
       path: /data
       type: Directory


Shell script 

services=("nginx" "docker" "httpd")

for service in "${services[@]}"; do
  if systemctl is-active --quiet "$service"; then
    echo "$service is running"
  else
    echo "$service is not running"
  fi
done
