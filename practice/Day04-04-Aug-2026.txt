Jenkinsfile — Basic CI Pipeline
pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building project'
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests'
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
    }
}



stages:
- build
- test 
- deploy 

build:
  stage:build 
  script:
    - echo "Building Project"

test:
   stage: test 
   script:
      - echo "Runing test"

deploy:
   stage: test
   script: 
      - echo "Deploy"
