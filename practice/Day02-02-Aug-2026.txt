

---
- name: install and configure nginx
  hosts: all
  become: true

  tasks:
    - name: install nginx web-server
      apt:
        name: nginx
        state: present
        update_cache: true

    - name: start and enable nginx web-server
      service:
        name: nginx
        state: started
        enabled: true

# Kubernetes — Deployment with 2 Replicas


apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: pathnex-app

  template:
    metadata:
      labels:
        app: pathnex-app

    spec:
      containers:
        - name: app
          image: nginx
          ports:
            - containerPort: 80
