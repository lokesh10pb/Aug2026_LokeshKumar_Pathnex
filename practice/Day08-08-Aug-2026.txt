#Ansible 

- name: Install dependencies
  hosts: all
  become: yes

  tasks:
    - name: Install python3-pip on Ubuntu/Debian
      apt:
        name: python3-pip
        state: present
        update_cache: yes
      when: ansible_os_family == "Debian"

    - name: Install python3-pip on RedHat/Amazon Linux
      yum:
        name: python3-pip
        state: present
      when: ansible_os_family == "RedHat"

    - name: Install docker-compose
      pip:
        name: docker-compose



#Terraform EC2 with Security Group and EBS Volume

name "aws_security_group" "allow_ssh_http"{
    name  = "allow_ssh_http"
    description = "Allow SSH and HTTP access"
    ingress{
         ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]

    }
    ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

resource "aws_instance" "pathnex_ec2"{
    ami = "ami-0abcd1234abcd1234"
    security_groups = [aws_security_group.allow_ssh_http.name]
    ebs_block_device{
        device_name = "dev/sdh"
        volume_size = 30
    }
}

apiVersion: v1
kind: Service
metadata:
  name: pathnex-service
spec:
  type: NodePort
  selector:
    app: pathnex-app
  ports:
    - protocol: TCP
      port: 80
      nodePort: 30007



pipeline {
    agent any
    stages {
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build('pathnex-app')
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image('pathnex-app').push('latest')
                    }
                }
            }
        }
    }
}
